### Task 2: Add The Minimal Gate Workflow

**Files:**
- Create: `.github/workflows/pr-gate.yml`

- [ ] **Step 1: Create the gate workflow**

Create `.github/workflows/pr-gate.yml` with this exact content:

```yaml
on:
  workflow_call:
    inputs:
      require-run-ci:
        description: "Whether the PR must have the run-ci label"
        type: boolean
        default: true

jobs:
  pr-gate:
    runs-on: ubuntu-latest
    steps:
      - name: Fetch latest PR info
        if: github.event_name == 'pull_request'
        id: pr
        uses: actions/github-script@v7
        with:
          github-token: ${{ secrets.GITHUB_TOKEN }}
          script: |
            const pr = await github.rest.pulls.get({
              owner: context.repo.owner,
              repo: context.repo.repo,
              pull_number: context.issue.number
            });
            core.setOutput("labels", JSON.stringify(pr.data.labels.map(l => l.name)));
            core.setOutput("draft", pr.data.draft);

      - name: Log PR info
        if: github.event_name == 'pull_request'
        run: |
          echo "PR Event: ${{ github.event_name }}"
          echo "PR Labels: ${{ steps.pr.outputs.labels }}"
          echo "PR Draft: ${{ steps.pr.outputs.draft }}"
          echo "Require run-ci: ${{ inputs.require-run-ci }}"

      - name: Block draft PR
        if: github.event_name == 'pull_request' && fromJson(steps.pr.outputs.draft)
        run: |
          echo "PR is draft. Blocking CI."
          exit 1

      - name: Require run-ci label
        if: github.event_name == 'pull_request' && inputs.require-run-ci == true
        run: |
          if [[ "${{ contains(fromJson(steps.pr.outputs.labels), 'run-ci') }}" == "false" ]]; then
            echo "Missing required label 'run-ci'."
            exit 1
          fi
```

Create `test/run_suite.py` with this exact content:

```python
#!/usr/bin/env python3
from __future__ import annotations

import argparse
import sys


VALID_SUITES = {"stage-a-dcu", "stage-b-dcu"}


def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser()
    parser.add_argument("--hw", required=True, choices=["dcu"])
    parser.add_argument("--suite", required=True, choices=sorted(VALID_SUITES))
    parser.add_argument("--auto-partition-id", type=int, default=None)
    parser.add_argument("--auto-partition-size", type=int, default=None)
    parser.add_argument("--continue-on-error", action="store_true")
    args = parser.parse_args()

    has_partition_id = args.auto_partition_id is not None
    has_partition_size = args.auto_partition_size is not None
    if has_partition_id != has_partition_size:
        parser.error("--auto-partition-id and --auto-partition-size must be specified together.")
    if args.auto_partition_size is not None:
        if args.auto_partition_size <= 0:
            parser.error("--auto-partition-size must be positive.")
        if not 0 <= args.auto_partition_id < args.auto_partition_size:
            parser.error(
                f"--auto-partition-id must be in range [0, {args.auto_partition_size}), "
                f"but got {args.auto_partition_id}"
            )
    return args


def main() -> int:
    args = parse_args()
    print("DCU mock suite execution")
    print(f"hw={args.hw}")
    print(f"suite={args.suite}")
    print(f"continue_on_error={args.continue_on_error}")
    if args.auto_partition_size is None:
        print("partition=none")
    else:
        print(
            f"partition_id={args.auto_partition_id} "
            f"partition_size={args.auto_partition_size}"
        )
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

### Task 4: Add The Minimal DCU PR Workflow

**Files:**
- Create: `.github/workflows/pr-test-dcu.yml`

- [ ] **Step 1: Create the minimal workflow**

Create `.github/workflows/pr-test-dcu.yml` with this exact content:

```yaml
name: PR Test (DCU)

run-name: ${{ inputs.target_stage && format('[{0}]', inputs.target_stage) || '' }}

on:
  pull_request:
    branches: [ main ]
    types: [opened, synchronize, reopened, labeled, ready_for_review]
    paths:
      - ".github/workflows/pr-test-dcu.yml"
      - ".github/workflows/pr-gate.yml"
      - "test/**"
  workflow_dispatch:
    inputs:
      target_stage:
        description: "Stage to run"
        required: true
        type: choice
        default: "stage-a-dcu"
        options:
          - stage-a-dcu
          - stage-b-dcu
      partition_size:
        description: "Number of stage-b partitions"
        required: false
        type: number
        default: 2
      continue_on_error:
        description: "Continue on error"
        required: false
        type: boolean
        default: false

concurrency:
  group: pr-test-dcu-${{ github.ref }}-${{ inputs.target_stage || 'all' }}
  cancel-in-progress: true

jobs:
  call-gate:
    uses: ./.github/workflows/pr-gate.yml
    secrets: inherit

  check-changes:
    needs: [call-gate]
    runs-on: ubuntu-latest
    outputs:
      dcu: ${{ steps.filter.outputs.dcu || steps.run-mode.outputs.run_all_tests }}
      run_stage_a: ${{ steps.stage.outputs.run_stage_a }}
      run_stage_b: ${{ steps.stage.outputs.run_stage_b }}
    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Determine run mode
        id: run-mode
        run: |
          if [[ "${{ github.event_name }}" == "workflow_dispatch" ]]; then
            echo "run_all_tests=true" >> "$GITHUB_OUTPUT"
          else
            echo "run_all_tests=false" >> "$GITHUB_OUTPUT"
          fi

      - name: Detect file changes
        id: filter
        uses: dorny/paths-filter@v3
        if: steps.run-mode.outputs.run_all_tests != 'true'
        with:
          filters: |
            dcu:
              - ".github/workflows/pr-test-dcu.yml"
              - ".github/workflows/pr-gate.yml"
              - "test/**"

      - name: Select stages
        id: stage
        run: |
          TARGET_STAGE="${{ inputs.target_stage }}"
          if [[ -z "$TARGET_STAGE" ]]; then
            echo "run_stage_a=true" >> "$GITHUB_OUTPUT"
            echo "run_stage_b=true" >> "$GITHUB_OUTPUT"
          elif [[ "$TARGET_STAGE" == "stage-a-dcu" ]]; then
            echo "run_stage_a=true" >> "$GITHUB_OUTPUT"
            echo "run_stage_b=false" >> "$GITHUB_OUTPUT"
          elif [[ "$TARGET_STAGE" == "stage-b-dcu" ]]; then
            echo "run_stage_a=false" >> "$GITHUB_OUTPUT"
            echo "run_stage_b=true" >> "$GITHUB_OUTPUT"
          else
            echo "Unsupported target stage: $TARGET_STAGE"
            exit 1
          fi

  stage-a-dcu:
    needs: [check-changes]
    if: needs.check-changes.outputs.dcu == 'true' && needs.check-changes.outputs.run_stage_a == 'true'
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Run mock stage-a DCU suite
        run: |
          cd test
          python3 run_suite.py --hw dcu --suite stage-a-dcu ${{ inputs.continue_on_error && '--continue-on-error' || '' }}

  stage-b-dcu:
    needs: [check-changes, stage-a-dcu]
    if: |
      always() &&
      needs.check-changes.outputs.dcu == 'true' &&
      needs.check-changes.outputs.run_stage_b == 'true' &&
      (needs.stage-a-dcu.result == 'success' || needs.stage-a-dcu.result == 'skipped')
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        partition: [0, 1]
    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Run mock stage-b DCU suite partition
        run: |
          cd test
          python3 run_suite.py \
            --hw dcu \
            --suite stage-b-dcu \
            --auto-partition-id ${{ matrix.partition }} \
            --auto-partition-size 2 \
            ${{ inputs.continue_on_error && '--continue-on-error' || '' }}
```

```yaml
              --hw dcu \
              --suite stage-a-dcu \
              --timeout-per-file 900 \
              ${{ needs.check-changes.outputs.continue_flag }}

  stage-b-dcu:
    needs: [check-changes, validate-config, stage-a-dcu]
    if: |
      always() &&
      needs.check-changes.outputs.dcu == 'true' &&
      needs.check-changes.outputs.run_stage_b == 'true' &&
      (needs.stage-a-dcu.result == 'success' || needs.stage-a-dcu.result == 'skipped')
    runs-on: ${{ needs.check-changes.outputs.runner_label }}
    strategy:
      fail-fast: false
      matrix:
        partition: [0, 1]
    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Setup DCU container
        run: bash scripts/ci/dcu/dcu_ci_start_container.sh --image "${{ needs.check-changes.outputs.image }}"

      - name: Install dependencies
        run: bash scripts/ci/dcu/dcu_ci_install_dependency.sh

      - name: Verify DCU registration
        run: |
          bash scripts/ci/dcu/dcu_ci_exec.sh -w /sglang-checkout \
            python3 scripts/ci/dcu/verify_dcu_registration.py

      - name: Run DCU stage-b suite partition
        timeout-minutes: 180
        run: |
          bash scripts/ci/dcu/dcu_ci_exec.sh -w /sglang-checkout/test \
            python3 run_suite.py \
              --hw dcu \
              --suite stage-b-dcu \
              --auto-partition-id ${{ matrix.partition }} \
              --auto-partition-size 2 \
              --timeout-per-file 900 \
              ${{ needs.check-changes.outputs.continue_flag }}
```
